function detectTitle(pid) {
  const p = pid.toLowerCase();
  if (/fantastic[\-_]four|ff_\d|f\._four/.test(p))           return 'Fantastic Four';
  if (/iron[\-_]man|ironman/.test(p))                         return 'Iron Man';
  if (/captain[\-_]america/.test(p))                          return 'Captain America';
  if (/journey[\-_]into[\-_]mystery|jim_\d/.test(p))         return 'Journey into Mystery';
  if (/tales[\-_]of[\-_]suspense|tos_\d|t\._o\._s/.test(p)) return 'Tales of Suspense';
  if (/tales[\-_]to[\-_]astonish|tta_\d|t\._t\._a/.test(p)) return 'Tales to Astonish';
  if (/daredevil/.test(p))                                    return 'Daredevil';
  if (/x[\-_]men|xmen/.test(p))                              return 'X-Men';
  if (/\bthor\b/.test(p))                                     return 'Thor';
  if (/avenger/.test(p))                                      return 'Avengers';
  if (/giant[\-_]size|g[\-_]size/.test(p))                   return 'Giant-Size';
  if (/wwbn|werewolf/.test(p))                                return 'Werewolf by Night';
  if (/asm_annual|spider[\-_]man[\-_]annual/.test(p))         return 'Amazing Spider-Man Annual';
  if (/asm_|amazing[\-_]spiderman|amazing[\-_]spider|spiderman_300/.test(p)) return 'Amazing Spider-Man';
  return 'Other';
}

function detectIssue(pid) {
  const clean = pid
    .replace(/cgc[\-_]?\d+[\.\-]\d+/gi,'')
    .replace(/pgx[\-_]?\d+[\.\-]\d+/gi,'')
    .replace(/ssx?\d*/gi,'')
    .replace(/front|cover|back|pedigree|canadian|variant/gi,'');
  const annual = clean.match(/annual[\-_](\d+)/i) || clean.match(/_ann[\-_](\d+)/i);
  if (annual) return 'Annual #' + annual[1];
  const m = clean.match(/[\-_](\d{1,4})[\-_]/) || clean.match(/[\-_](\d{1,4})$/);
  if (m && parseInt(m[1]) < 2000) return '#' + parseInt(m[1]);
  return '';
}

function detectGrade(pid) {
  const m = pid.match(/cgc[\-_\s]?(\d+\.\d)/i) || pid.match(/[\-_](\d+\.\d)[\-_]/);
  return m ? m[1] : '?';
}

function detectSig(pid) {
  const p = pid.toLowerCase();
  if (/ssx4|ss_x4/.test(p)) return 'SS ×4';
  if (/ssx3|ss_x3/.test(p)) return 'SS ×3';
  if (/ssx2|ss_x2/.test(p)) return 'SS ×2';
  if (/_ss[\-_]|_ss$|[\-_]ss[\-_]/.test(p)) return 'SS';
  return null;
}

function comicCard(c) {
  const flagClass = c.sig ? ' ss' : '';

  return '<div class="comic-card">'

let ALL_COMICS = [];

function renderCatalog(filter) {
  const container = document.getElementById('catalogContainer');
  const filtered  = filter === 'ss' ? ALL_COMICS.filter(c => c.sig) : ALL_COMICS;
  const groups = {};
  filtered.forEach(c => { if (!groups[c.title]) groups[c.title] = []; groups[c.title].push(c); });
  const extras   = Object.keys(groups).filter(t => !TITLE_ORDER.includes(t)).sort();
  const sections = [...TITLE_ORDER, ...extras];
  container.innerHTML = sections.map(title => {
    const comics = groups[title] || [];
    const count  = comics.length;
    const gid    = title.replace(/[^a-z0-9]/gi,'_');
    const badge  = count > 0 ? '<div class="title-group-count">' + count + ' ' + (count===1?'copy':'copies') + '</div>' : '';
    const body   = count > 0
      ? '<div class="comic-grid">' + comics.map(comicCard).join('') + '</div>'
      : '<div class="title-group-empty">No copies currently listed — check back soon.</div>';
    return '<div class="title-group" id="group_' + gid + '">'
      + '<div class="title-group-header"><div class="title-group-name">' + title + '</div>' + badge + '</div>'
      + body + '</div>';
  }).join('');
}

    if (cursor) url += '&next_cursor=' + encodeURIComponent(cursor);
    const resp = await fetch(url, {
      headers: { 'Authorization': 'Basic ' + btoa(KEY + ':' + SEC) }
    });
    if (!resp.ok) throw new Error('API error ' + resp.status);
    const data = await resp.json();
    (data.resources || []).forEach(r => {
      if (!seen.has(r.public_id)) { seen.add(r.public_id); assets.push(r); }
    });
    cursor = data.next_cursor || null;
  } while (cursor);
  return assets;
}

async function init() {
  try {
    const assets = await fetchAllAssets();
    ALL_COMICS = assets.map(a => ({
      title: detectTitle(a.public_id),
      issue: detectIssue(a.public_id),
      grade: detectGrade(a.public_id),
      sig:   detectSig(a.public_id),
      img:   a.secure_url,
      pid:   a.public_id
    })).sort((a, b) => {
      const ti = t => { const i = TITLE_ORDER.indexOf(t); return i < 0 ? 999 : i; };
      const td = ti(a.title) - ti(b.title);
      if (td !== 0) return td;
      return (parseInt((a.issue||'').replace(/\D/g,''))||0) - (parseInt((b.issue||'').replace(/\D/g,''))||0);
    });
    const total   = ALL_COMICS.length;
    const nTitles = [...new Set(ALL_COMICS.map(c => c.title))].length;
    document.getElementById('heroTotal').textContent   = total;
    document.getElementById('heroTitles').textContent  = nTitles;
    document.getElementById('bundleTotal').textContent = total;
    document.getElementById('footerNote').textContent  = total + ' GRADED COPIES · ' + nTitles + ' TITLE GROUPS';
    renderCatalog('all');
  } catch(err) {
    document.getElementById('catalogContainer').innerHTML =
      '<div class="err-msg">Could not load collection.<br><br>' + err.message + '</div>';
  }
}
